The images above use images.unsplash.com, which should work reliably for the fitness website aesthetic, but they are generic fitness/supplement-related images, not necessarily the exact product packaging.

For exact product images from Google, the reliable method is:

Download the image.
Create an images folder beside index.html.
Put your downloaded product images there.
Use:
<img src="images/avvatar-alpha-whey-1kg.png">

That will be much more reliable than directly copying a Google Images URL.

// Reliable product-image loader: proxies blocked retailer/CDN images and prevents empty cards.
const productImages = document.querySelectorAll(".product-visual img, .hero-product img, #modalImage");
function proxiedImage(url){
  if(!url || url.startsWith("data:")) return url;
  if(url.includes("images.weserv.nl")) return url;
  return "https://images.weserv.nl/?url=" + encodeURIComponent(url.replace(/^https?:\/\//,"")) + "&w=900&output=webp&q=92";
}
productImages.forEach(img=>{
  const original = img.getAttribute("src");
  if(original) img.src = proxiedImage(original);
  img.addEventListener("error",()=>{
    const backup = img.dataset.backup;
    if(backup){ img.src = proxiedImage(backup); return; }
    img.style.background = "#111";
  },{once:true});
});
